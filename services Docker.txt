services:
  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_PASSWORD: [cek vault, jangan hard-code]
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - internal

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    networks:
      - internal

  minio:
    image: minio/minio:latest
    environment:
      MINIO_ROOT_USER: [dari vault]
      MINIO_ROOT_PASSWORD: [dari vault]
    command: server /data --console-address ":9001"
    volumes:
      - minio_data:/data
    networks:
      - internal

  api:
    build: ./backend
    depends_on:
      - postgres
      - redis
      - minio
    environment:
      DATABASE_URL: postgres://user:pass@postgres:5432/reports
      REDIS_URL: redis://redis:6379
      MINIO_URL: http://minio:9000
    labels:
      - "traefik.http.routers.api.rule=Host(`api.domain.com`)"
    networks:
      - web
      - internal

  worker:
    build: ./backend
    command: node dist/worker/main.js
    depends_on:
      - postgres
      - redis
    environment:
      DATABASE_URL: postgres://user:pass@postgres:5432/reports
      REDIS_URL: redis://redis:6379
    networks:
      - internal

volumes:
  postgres_data:
  redis_data:
  minio_data:

networks:
  web:
    external: true
  internal:
    internal: true
